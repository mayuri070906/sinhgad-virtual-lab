💻 C Implementation

#define MAX 5

int stack[MAX];
int top = -1;

void push(int value)
{
    if(top == MAX - 1)
    {
        printf("Stack Overflow");
    }
    else
    {
        stack[++top] = value;
    }
}

void pop()
{
    if(top == -1)
    {
        printf("Stack Underflow");
    }
    else
    {
        top--;
    }
}
