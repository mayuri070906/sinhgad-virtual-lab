💻 C Implementation

#define MAX 5

int queue[MAX];

int front = -1;
int rear = -1;

void enqueue(int value)
{
    if(rear == MAX - 1)
    {
        printf("Queue Overflow");
    }
    else
    {
        if(front == -1)
            front = 0;

        queue[++rear] = value;
    }
}

void dequeue()
{
    if(front == -1 || front > rear)
    {
        printf("Queue Underflow");
    }
    else
    {
        front++;
    }
}
