📖 What is Doubly Linked List?

A Doubly Linked List is a linear data structure in which every node contains three parts: a pointer to the previous node, data and a pointer to the next node.

The previous pointer stores the address of the previous node, while the next pointer stores the address of the next node.

Key Idea: Previous Pointer + Data + Next Pointer

🧩 Structure of Node

Each node of a Doubly Linked List contains three components:

  • Prev: Stores the address of the previous node.
  • Data: Stores the actual value.
  • Next: Stores the address of the next node. 
struct Node
{
    struct Node *prev;

    int data;

    struct Node *next;
};

🔗 Example

A Doubly Linked List containing three nodes can be represented as:

NULL ⇄ 10 ⇄ 20 ⇄ 30 ⇄ NULL

The first node has prev = NULL and the last node has next = NULL.

⚙️ Operations

The following operations can be performed on a Doubly Linked List:

  • Begin Insert – Insert a node at the beginning.
  • End Insert – Insert a node at the end.
  • Location Insert – Insert a node at a specific position.
  • Begin Delete – Delete the first node.
  • End Delete – Delete the last node.
  • Location Delete – Delete a node from a specific position.
  • Search – Search for a particular value.

✅ Advantages

  • Supports forward and backward traversal.
  • Deletion of a node is easier because the previous node is directly available.
  • Insertion and deletion can be performed at both ends.
  • Useful when two-way navigation is required.

❌ Disadvantages

  • Requires extra memory for the previous pointer.
  • More complex than a Singly Linked List.
  • More pointer manipulation is required.

📱 Applications

  • Browser forward and backward navigation.
  • Undo and redo operations.
  • Music playlist navigation.
  • Image viewer navigation.
  • Navigation systems.

⏱ Time Complexity

Operation Time Complexity
Begin Insert O(1)
End Insert O(n)
Location Insert O(n)
Begin Delete O(1)
End Delete O(n)
Location Delete O(n)
Search O(n)

💻 C Language Implementation

Complete implementation of Doubly Linked List using C language. 

#include <stdio.h>
#include <stdlib.h>

struct Node
{
    struct Node *prev;
    int data;
    struct Node *next;
};

struct Node *head = NULL;


/* Insert at Beginning */

void insertBegin(int value)
{
    struct Node *newNode;

    newNode = (struct Node*)malloc(sizeof(struct Node));

    newNode->data = value;
    newNode->prev = NULL;
    newNode->next = head;

    if (head != NULL)
    {
        head->prev = newNode;
    }

    head = newNode;
}


/* Insert at End */

void insertEnd(int value)
{
    struct Node *newNode;
    struct Node *temp;

    newNode = (struct Node*)malloc(sizeof(struct Node));

    newNode->data = value;
    newNode->next = NULL;

    if (head == NULL)
    {
        newNode->prev = NULL;
        head = newNode;
        return;
    }

    temp = head;

    while (temp->next != NULL)
    {
        temp = temp->next;
    }

    temp->next = newNode;
    newNode->prev = temp;
}


/* Insert at Location */

void insertLocation(int value, int position)
{
    struct Node *newNode;
    struct Node *temp;

    if (position == 1)
    {
        insertBegin(value);
        return;
    }

    newNode = (struct Node*)malloc(sizeof(struct Node));

    newNode->data = value;

    temp = head;

    for (int i = 1;
         i < position - 1 && temp != NULL;
         i++)
    {
        temp = temp->next;
    }

    if (temp == NULL)
    {
        printf("Invalid position\n");
        free(newNode);
        return;
    }

    newNode->next = temp->next;
    newNode->prev = temp;

    if (temp->next != NULL)
    {
        temp->next->prev = newNode;
    }

    temp->next = newNode;
}


/* Delete at Beginning */

void deleteBegin()
{
    struct Node *temp;

    if (head == NULL)
    {
        printf("List is empty\n");
        return;
    }

    temp = head;

    head = head->next;

    if (head != NULL)
    {
        head->prev = NULL;
    }

    free(temp);
}


/* Delete at End */

void deleteEnd()
{
    struct Node *temp;

    if (head == NULL)
    {
        printf("List is empty\n");
        return;
    }

    temp = head;

    while (temp->next != NULL)
    {
        temp = temp->next;
    }

    if (temp->prev != NULL)
    {
        temp->prev->next = NULL;
    }
    else
    {
        head = NULL;
    }

    free(temp);
}


/* Delete at Location */

void deleteLocation(int position)
{
    struct Node *temp;

    if (head == NULL)
    {
        printf("List is empty\n");
        return;
    }

    temp = head;

    for (int i = 1;
         i < position && temp != NULL;
         i++)
    {
        temp = temp->next;
    }

    if (temp == NULL)
    {
        printf("Invalid position\n");
        return;
    }

    if (temp->prev != NULL)
    {
        temp->prev->next = temp->next;
    }
    else
    {
        head = temp->next;
    }

    if (temp->next != NULL)
    {
        temp->next->prev = temp->prev;
    }

    free(temp);
}


/* Search */

void search(int value)
{
    struct Node *temp = head;

    int position = 1;

    while (temp != NULL)
    {
        if (temp->data == value)
        {
            printf("Found at position %d\n", position);
            return;
        }

        temp = temp->next;
        position++;
    }

    printf("Value not found\n");
}


/* Display Forward */

void displayForward()
{
    struct Node *temp = head;

    while (temp != NULL)
    {
        printf("%d <-> ", temp->data);
        temp = temp->next;
    }

    printf("NULL\n");
}


/* Display Backward */

void displayBackward()
{
    struct Node *temp = head;

    if (temp == NULL)
    {
        printf("List is empty\n");
        return;
    }

    while (temp->next != NULL)
    {
        temp = temp->next;
    }

    while (temp != NULL)
    {
        printf("%d <-> ", temp->data);
        temp = temp->prev;
    }

    printf("NULL\n");
}


/* Main */

int main()
{
    insertBegin(10);

    insertEnd(20);

    insertEnd(30);

    insertLocation(15, 2);

    printf("Forward:\n");
    displayForward();

    printf("Backward:\n");
    displayBackward();

    search(20);

    deleteBegin();

    deleteEnd();

    deleteLocation(2);

    displayForward();

    return 0;
}
